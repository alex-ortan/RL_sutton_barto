\begin{frame}{$\epsilon$-greedy agent}
\begin{columns}

    \begin{column}{0.5\textwidth}
        \begin{block}{Observation 1:}
        For $\epsilon=0$, average reward $\approx 1$
        \end{block}
        \begin{block}{What gives?}
            \footnotesize
            \begin{itemize}[label=$\rightarrow$]
                \item This is the expected value of the first positive reward received.
                \item Can be estimated via integrals weighted by survival functions - the chance that a random walk stays positive forever.
                \item Depends on initial action value estimates ($0$ here).
                \item Value is linear in the variance of the expected reward over actions.
            \end{itemize}
        \end{block}
    \end{column}

    \begin{column}{0.5\textwidth}
        \centering
        \includegraphics[width=\textwidth]{figures/fig2-2.png}
    \end{column}

\end{columns}

\begin{frame}{$\epsilon$-greedy agent}
\begin{columns}

    \begin{column}{0.5\textwidth}
        \begin{block}{Observation 2:}
        Average reward $\lesssim 1.54$
        \end{block}
        \begin{block}{What gives?}
            \footnotesize
            \begin{itemize}[label=$\rightarrow$]
                \item Theoretical ceiling, as provided by an oracle agent.
                \item This is the best \textbf{any} agent can do.
                \item Calculated as the expected value of the highest of $k$ action values.
                \item Ceiling $\approx \sqrt{2 \ln k}$
                \item For $k=10$, this ceiling $\approx 1.539$.
            \end{itemize}
        \end{block}
    
    \end{column}

\begin{frame}{Constant Step-size Estimator}
\begin{columns}

    \begin{column}{0.5\textwidth}
        \begin{block}{Observations}
        \footnotesize
        \begin{itemize}[label=$\rightarrow$]
            \item Sample Average is not an effective estimator for a non-stationary problem because it places 
                equal weight on old rewards which may no longer reflect current action values.
            \item The constant step-size estimator is much better at tracking correct action values because it
                exponentially discounts old rewards.
            \item Note that this approach is not efficient in the stationary case.
        \end{itemize}
        \end{block}
    \end{column}

\begin{frame}{Constant Step-size Estimator and Optimistic Initial Values}
\begin{columns}

    \begin{column}{0.5\textwidth}
        \begin{block}{Observations}
        \footnotesize
        \begin{itemize}[label=$\rightarrow$]
            \item For the stationary problem, constant step-size is less efficient than sample average.
            \item Using optimistic initial values with constant step-size estimates encourages a lot of early 
                exploration and helps overcome the initial inefficiency, eventually performing better than the 
                sample average estimates.
            \item Note that using optimistic values with $\epsilon = 0$ would be counterproductive, as the two 
                mechanisms would work against each other in the early steps.
        \end{itemize}
        \end{block}
    \end{column}

\begin{frame}{Upper-Confidence-Bound Action Selection (UCB)}
\begin{columns}

    \begin{column}{0.5\textwidth}
        \begin{block}{Observations}
        \begin{itemize}[label=$\rightarrow$]
            \item UCB performs better than $\epsilon$-greedy, using the sample average estimator.
            \item UCB also outperforms the greedy constant step-size with optimistic initial values.
            \item Does not perform very well on non-stationary problems, since the $1/\sqrt{n}$ does not 
                accurately reflect estimate uncertainty in the non-stationary case.
        \end{itemize}
        \end{block}
    \end{column}
